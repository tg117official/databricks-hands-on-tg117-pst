1. A customer record changes

A dimension contains descriptive business data. Customer city, email and membership tier can change after the first load. The warehouse must decide what to do with the old value.

Initial snapshot10 customers loaded on 1 January 2026.
Incremental Batch 18 incoming rows on 1 February 2026.
Incremental Batch 28 incoming rows on 1 March 2026.
Business keycustomer_id identifies the same customer across every batch.
Surrogate keycustomer_key is generated inside the warehouse. Type 2 creates a new key for a new version.
Changed rowThe business key already exists, but at least one tracked attribute is different.
Idempotent rerunRunning the same batch again must not create another change or duplicate row.
Scope: the incoming rows are already available as incremental data. This session does not repeat CDC extraction. It focuses on applying the rows to dimensions.

MYSQL_PORT = "3306"
MYSQL_DATABASE = "scd_customer_demo"
MYSQL_USER = "root"
MYSQL_PASSWORD = getpass("Enter the local MySQL password: ")
MYSQL_DRIVER = "com.mysql.cj.jdbc.Driver"

JDBC_URL = (
    f"jdbc:mysql://{MYSQL_HOST}:{MYSQL_PORT}/{MYSQL_DATABASE}"
    "?useSSL=false"
    "&allowPublicKeyRetrieval=true"
    "&serverTimezone=UTC"
)

def read_mysql(table_name):
    return (
        spark.read
        .format("jdbc")
        .option("url", JDBC_URL)
        .option("dbtable", table_name)
        .option("user", MYSQL_USER)
        .option("password", MYSQL_PASSWORD)
        .option("driver", MYSQL_DRIVER)
        .load()
    )

def write_mysql_append(dataframe, table_name):
    (
        dataframe.coalesce(1)
        .write
        .format("jdbc")
        .option("url", JDBC_URL)
        .option("dbtable", table_name)
        .option("user", MYSQL_USER)
        .option("password", MYSQL_PASSWORD)
        .option("driver", MYSQL_DRIVER)
        .option("batchsize", "1000")
        .mode("append")
        .save()
    )

def write_mysql_stage(dataframe, table_name):
    (
        dataframe.coalesce(1)
        .write
        .format("jdbc")
        .option("url", JDBC_URL)
        .option("dbtable", table_name)
        .option("user", MYSQL_USER)
        .option("password", MYSQL_PASSWORD)
        .option("driver", MYSQL_DRIVER)
        .option("batchsize", "1000")
        .mode("overwrite")
        .save()
    )

def execute_mysql_transaction(sql_statements):
    jvm = spark.sparkContext._gateway.jvm
    jvm.java.lang.Class.forName(MYSQL_DRIVER)

    connection = None
    statement = None

    try:
        connection = jvm.java.sql.DriverManager.getConnection(
            JDBC_URL,
            MYSQL_USER,
            MYSQL_PASSWORD
        )
        connection.setAutoCommit(False)
        statement = connection.createStatement()

        for sql_statement in sql_statements:
            statement.executeUpdate(sql_statement)

        connection.commit()
    except Exception:
        if connection is not None:
            connection.rollback()
        raise
    finally:
        if statement is not None:
            statement.close()
        if connection is not None:
            connection.close()

def different_value(left_alias, right_alias, columns):
    expression = F.lit(False)

    for column_name in columns:
        expression = expression | (
            ~F.col(f"{left_alias}.{column_name}").eqNullSafe(
                F.col(f"{right_alias}.{column_name}")
            )
        )

    return expression

def validate_incoming_batch(dataframe, batch_name, expected_rows):
    duplicate_keys = (
        dataframe
        .groupBy("customer_id")
        .count()
        .filter(F.col("count") > 1)
        .count()
    )

    null_required_values = (
        dataframe
        .filter(
            F.col("customer_id").isNull()
            | F.col("customer_name").isNull()
            | F.col("city").isNull()
            | F.col("email").isNull()
            | F.col("membership_tier").isNull()
            | F.col("updated_at").isNull()
        )
        .count()
    )

    actual_rows = dataframe.count()

    assert actual_rows == expected_rows, (
        f"{batch_name}: expected {expected_rows} rows, "
        f"but found {actual_rows}"
    )
    assert duplicate_keys == 0, f"{batch_name}: duplicate customer_id found"
    assert null_required_values == 0, f"{batch_name}: required value is missing"

    print(batch_name, "PASS")

7. SCD Type 1: apply two ingestion batches

Type 1 keeps only the latest values. A changed customer is updated in place, so the existing surrogate key remains the same and the old attribute values are lost.

How the code works

  1. Read the current Type 1 dimension.
  2. Classify each incoming row as NEW, CHANGED or UNCHANGED.
  3. Write only new and changed rows to staging.
  4. Insert new customers and update existing customers using the business key.
Run 5A in the PySpark shell - define the SCD Type 1 function
PySpark
TYPE1_TRACKED_COLUMNS = [
    "customer_name",
    "city",
    "email",
    "membership_tier"
]

def apply_scd_type1(incoming_df, batch_name):
    current_df = read_mysql("dim_customer_type1")

    classified_df = (
        incoming_df.alias("i")
        .join(
            current_df.alias("d"),
            F.col("i.customer_id") == F.col("d.customer_id"),
            "left"
        )
        .select(
            F.col("i.customer_id"),
            F.col("i.customer_name"),
            F.col("i.city"),
            F.col("i.email"),
            F.col("i.membership_tier"),
            F.col("i.updated_at"),
            F.when(
                F.col("d.customer_id").isNull(),
                F.lit("NEW")
            ).when(
                different_value("i", "d", TYPE1_TRACKED_COLUMNS),
                F.lit("CHANGED")
            ).otherwise(F.lit("UNCHANGED")).alias("action")
        )
    )

    print(f"Type 1 - {batch_name}")
    classified_df.groupBy("action").count().orderBy("action").show()

    rows_to_apply_df = (
        classified_df
        .filter(F.col("action").isin("NEW", "CHANGED"))
        .drop("action")
    )

    rows_to_apply = rows_to_apply_df.count()

    if rows_to_apply == 0:
        print("No Type 1 rows need to be applied.")
        return classified_df

    write_mysql_stage(rows_to_apply_df, "stg_customer_type1")

    execute_mysql_transaction([
        """
        UPDATE dim_customer_type1 AS destination
        INNER JOIN stg_customer_type1 AS incoming
            ON destination.customer_id = incoming.customer_id
        SET
            destination.customer_name = incoming.customer_name,
            destination.city = incoming.city,
            destination.email = incoming.email,
            destination.membership_tier = incoming.membership_tier,
            destination.updated_at = incoming.updated_at
        """,
        """
        INSERT INTO dim_customer_type1
        (
            customer_id,
            customer_name,
            city,
            email,
            membership_tier,
            updated_at
        )
        SELECT
            incoming.customer_id,
            incoming.customer_name,
            incoming.city,
            incoming.email,
            incoming.membership_tier,
            incoming.updated_at
        FROM stg_customer_type1 AS incoming
        LEFT JOIN dim_customer_type1 AS destination
            ON incoming.customer_id = destination.customer_id
        WHERE destination.customer_id IS NULL
        """
    ])

    print("Rows applied:", rows_to_apply)
    return classified_df

8. SCD Type 2: apply two ingestion batches

Type 2 keeps complete history. When a tracked attribute changes, the current row is expired and a new current version is inserted with a new surrogate key.

effective_fromThe first date on which this version is valid.
effective_toThe last date on which an expired version was valid. Current rows use NULL.
is_current1 identifies the current version and 0 identifies an old version.
version_numberStarts at 1 and increases for every new version of the same customer.
Run 6A in the PySpark shell - define the SCD Type 2 function
PySpark
TYPE2_TRACKED_COLUMNS = [
    "customer_name",
    "city",
    "email",
    "membership_tier"
]

def apply_scd_type2(incoming_df, batch_name):
    current_df = (
        read_mysql("dim_customer_type2")
        .filter(F.col("is_current") == 1)
    )

    classified_df = (
        incoming_df.alias("i")
        .join(
            current_df.alias("d"),
            F.col("i.customer_id") == F.col("d.customer_id"),
            "left"
        )
        .select(
            F.col("i.customer_id"),
            F.col("i.customer_name"),
            F.col("i.city"),
            F.col("i.email"),
            F.col("i.membership_tier"),
            F.col("i.updated_at").alias("effective_from"),
            F.lit(None).cast("date").alias("effective_to"),
            F.lit(1).cast("int").alias("is_current"),
            F.when(
                F.col("d.customer_id").isNull(),
                F.lit(1)
            ).otherwise(
                F.col("d.version_number") + F.lit(1)
            ).cast("int").alias("version_number"),
            F.when(
                F.col("d.customer_id").isNull(),
                F.lit("NEW")
            ).when(
                different_value("i", "d", TYPE2_TRACKED_COLUMNS),
                F.lit("CHANGED")
            ).otherwise(F.lit("UNCHANGED")).alias("action")
        )
    )

    print(f"Type 2 - {batch_name}")
    classified_df.groupBy("action").count().orderBy("action").show()

    rows_to_apply_df = classified_df.filter(
        F.col("action").isin("NEW", "CHANGED")
    )

    rows_to_apply = rows_to_apply_df.count()

    if rows_to_apply == 0:
        print("No Type 2 rows need to be applied.")
        return classified_df

    write_mysql_stage(rows_to_apply_df, "stg_customer_type2")

    execute_mysql_transaction([
        """
        UPDATE dim_customer_type2 AS destination
        INNER JOIN stg_customer_type2 AS incoming
            ON destination.customer_id = incoming.customer_id
        SET
            destination.effective_to = DATE_SUB(
                incoming.effective_from,
                INTERVAL 1 DAY
            ),
            destination.is_current = 0
        WHERE destination.is_current = 1
          AND incoming.action = 'CHANGED'
        """,
        """
        INSERT INTO dim_customer_type2
        (
            customer_id,
            customer_name,
            city,
            email,
            membership_tier,
            effective_from,
            effective_to,
            is_current,
            version_number
        )
        SELECT
            customer_id,
            customer_name,
            city,
            email,
            membership_tier,
            effective_from,
            effective_to,
            is_current,
            version_number
        FROM stg_customer_type2
        WHERE action IN ('NEW', 'CHANGED')
        """
    ])

    print("New Type 2 versions inserted:", rows_to_apply)
    return classified_df

9. SCD Type 3: apply two ingestion batches

Type 3 keeps the current city and one previous city in the same row. Other attributes in this example keep only their latest values.

City rule

  • If the city changes, copy current_city to previous_city.
  • Store the incoming city as the new current_city.
  • If only email or membership changes, keep both city columns unchanged.
Run 7A in the PySpark shell - define the SCD Type 3 function
PySpark
def apply_scd_type3(incoming_df, batch_name):
    current_df = read_mysql("dim_customer_type3")

    city_changed = ~F.col("i.city").eqNullSafe(
        F.col("d.current_city")
    )

    any_value_changed = (
        ~F.col("i.customer_name").eqNullSafe(F.col("d.customer_name"))
        | city_changed
        | ~F.col("i.email").eqNullSafe(F.col("d.email"))
        | ~F.col("i.membership_tier").eqNullSafe(
            F.col("d.membership_tier")
        )
    )

    classified_df = (
        incoming_df.alias("i")
        .join(
            current_df.alias("d"),
            F.col("i.customer_id") == F.col("d.customer_id"),
            "left"
        )
        .select(
            F.col("i.customer_id"),
            F.col("i.customer_name"),
            F.col("i.city").alias("current_city"),
            F.when(
                F.col("d.customer_id").isNull(),
                F.lit(None).cast("string")
            ).when(
                city_changed,
                F.col("d.current_city")
            ).otherwise(
                F.col("d.previous_city")
            ).alias("previous_city"),
            F.col("i.email"),
            F.col("i.membership_tier"),
            F.col("i.updated_at"),
            F.when(
                F.col("d.customer_id").isNull(),
                F.lit("NEW")
            ).when(
                any_value_changed,
                F.lit("CHANGED")
            ).otherwise(F.lit("UNCHANGED")).alias("action")
        )
    )

    print(f"Type 3 - {batch_name}")
    classified_df.groupBy("action").count().orderBy("action").show()

    rows_to_apply_df = (
        classified_df
        .filter(F.col("action").isin("NEW", "CHANGED"))
        .drop("action")
    )

    rows_to_apply = rows_to_apply_df.count()

    if rows_to_apply == 0:
        print("No Type 3 rows need to be applied.")
        return classified_df

    write_mysql_stage(rows_to_apply_df, "stg_customer_type3")

    execute_mysql_transaction([
        """
        UPDATE dim_customer_type3 AS destination
        INNER JOIN stg_customer_type3 AS incoming
            ON destination.customer_id = incoming.customer_id
        SET
            destination.customer_name = incoming.customer_name,
            destination.current_city = incoming.current_city,
            destination.previous_city = incoming.previous_city,
            destination.email = incoming.email,
            destination.membership_tier = incoming.membership_tier,
            destination.updated_at = incoming.updated_at
        """,
        """
        INSERT INTO dim_customer_type3
        (
            customer_id,
            customer_name,
            current_city,
            previous_city,
            email,
            membership_tier,
            updated_at
        )
        SELECT
            incoming.customer_id,
            incoming.customer_name,
            incoming.current_city,
            incoming.previous_city,
            incoming.email,
            incoming.membership_tier,
            incoming.updated_at
        FROM stg_customer_type3 AS incoming
        LEFT JOIN dim_customer_type3 AS destination
            ON incoming.customer_id = destination.customer_id
        WHERE destination.customer_id IS NULL
        """
    ])

    print("Rows applied:", rows_to_apply)
    return classified_df
